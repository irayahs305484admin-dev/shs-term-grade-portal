function doPost(e) { try { const requestData = JSON.parse(e.postData.contents); const username = requestData.username ? requestData.username.trim() : ""; const password = requestData.password ? requestData.password.trim() : ""; const ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Authenticate Student const studentsSheet = ss.getSheetByName("Students"); const studentsData = studentsSheet.getDataRange().getValues(); let student = null; for (let i = 1; i < studentsData.length; i++) { if (String(studentsData[i][1]) === username && String(studentsData[i][2]) === password) { student = { studentId: String(studentsData[i][0]), username: studentsData[i][1], fullName: studentsData[i][3], gradeLevel: parseInt(studentsData[i][4], 10) }; break; } } if (!student) { return responseJSON({ status: "error", message: "Invalid Username or Password" }); } // 2. Fetch Grades & Calculate General Average let gradeReport = []; let generalAverage = null; if (student.gradeLevel === 11) { const g11Sheet = ss.getSheetByName("Grade11"); const g11Data = g11Sheet.getDataRange().getValues(); let totalWeightedScore = 0; let totalUnits = 0; for (let i = 1; i < g11Data.length; i++) { if (String(g11Data[i][0]) === student.studentId) { const type = g11Data[i][1]; const subject = g11Data[i][2]; const t1 = parseFloat(g11Data[i][3]) || null; const t2 = parseFloat(g11Data[i][4]) || null; const t3 = parseFloat(g11Data[i][5]) || null; let finalGrade = null; let units = (type === "Core") ? 6 : 3; if (type === "Core") { const validTerms = [t1, t2, t3].filter(v => v !== null); if (validTerms.length > 0) { const sum = validTerms.reduce((acc, curr) => acc + curr, 0); finalGrade = Math.round(sum / validTerms.length); } } else if (type === "Elective") { const termVal = [t1, t2, t3].find(v => v !== null); if (termVal !== null && termVal !== undefined) { finalGrade = Math.round(termVal); } } // Accumulate weighted points if a final grade exists if (finalGrade !== null) { totalWeightedScore += finalGrade * units; totalUnits += units; } gradeReport.push({ subjectType: type, subjectName: subject, units: units, term1: t1, term2: t2, term3: t3, finalRating: finalGrade }); } } // Calculate General Weighted Average rounded to nearest whole number if (totalUnits > 0) { generalAverage = Math.round(totalWeightedScore / totalUnits); } } else if (student.gradeLevel === 12) { const g12Sheet = ss.getSheetByName("Grade12"); const g12Data = g12Sheet.getDataRange().getValues(); for (let i = 1; i < g12Data.length; i++) { if (String(g12Data[i][0]) === student.studentId) { const term = g12Data[i][1]; const subject = g12Data[i][2]; const gradeVal = parseFloat(g12Data[i][3]); gradeReport.push({ term: term, subjectName: subject, finalRating: isNaN(gradeVal) ? null : Math.round(gradeVal) }); } } } return responseJSON({ status: "success", student: student, grades: gradeReport, generalAverage: generalAverage }); } catch (err) { return responseJSON({ status: "error", message: err.toString() }); } } function doGet(e) { return ContentService.createTextOutput(JSON.stringify({ status: "online", message: "SHS Grade Portal API is active." })).setMimeType(ContentService.MimeType.JSON); } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
